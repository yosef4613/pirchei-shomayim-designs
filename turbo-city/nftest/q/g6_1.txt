// position + tangent + sample index at arc length s (s may be negative: extrapolates backwards along the first tangent)
function apttAt(P,s){

}
// trailer tracking: the main gear follows the nose gear at a rigid distance d (aircraft off-tracking in turns)
function apttTrack(P,d,th0){

}

// ---------- speed profiles: v(s) sampled every 3 px, time table by trapezoid integration ------------------------------------------------------
function apttProf(len,fn){
 const n=Math.max(2,Math.ceil(len/3)),d=len/n,v=new Float64Array(n+1),T=new Float64Array(n+1);
 for(let i=0;i<=n;i++)v[i]=Math.max(3,fn(i*d));
 for(let i=1;i<=n;i++)T[i]=T[i-1]+2*d/(v[i-1]+v[i]);
 return{len,n,d,v,T,dur:T[n]};
}
function apttTof(p,s){if(s<=0)return 0;if(s>=p.len)return p.dur;const f=s/p.d,i=Math.min(p.n-1,Math.floor(f));return p.T[i]+(p.T[i+1]-p.T[i])*(f-i)}
function apttSof(p,t){
 if(t<=0)return 0;if(t>=p.dur)return p.len;
 let lo=0,hi=p.n;while(hi-lo>1){const m=(lo+hi)>>1;if(p.T[m]<=t)lo=m;else hi=m}
 return(lo+(t-p.T[lo])/(p.T[hi]-p.T[lo]))*p.d;
}
// generic ground profile: start/end speeds, accelerations, slow zones (arcs) with a smooth ramp
function apttVfn(len,o){

  return v};
}

function apttObbHit(A,B,m){
